cmake_minimum_required(VERSION 3.20)
project(slang_mlir LANGUAGES CXX)
find_package(MLIR REQUIRED CONFIG)
find_package(LLVM REQUIRED CONFIG)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include_directories(${MLIR_INCLUDE_DIRS})
include_directories(${LLVM_INCLUDE_DIRS})
include_directories(${CMAKE_CURRENT_BINARY_DIR}/include)
include_directories(${CMAKE_CURRENT_SOURCE_DIR}/include)
add_definitions(${MLIR_DEFINITIONS})

set(LLVM_TARGET_DEFINITIONS ${CMAKE_CURRENT_SOURCE_DIR}/include/slang/IR/SlangOps.td)
mlir_tablegen(${CMAKE_CURRENT_BINARY_DIR}/include/slang/IR/SlangDialect.h.inc -gen-dialect-decls -dialect=slang)
mlir_tablegen(${CMAKE_CURRENT_BINARY_DIR}/include/slang/IR/SlangDialect.cpp.inc -gen-dialect-defs -dialect=slang)
mlir_tablegen(${CMAKE_CURRENT_BINARY_DIR}/include/slang/IR/SlangOps.h.inc -gen-op-decls -name slang)
mlir_tablegen(${CMAKE_CURRENT_BINARY_DIR}/include/slang/IR/SlangOps.cpp.inc -gen-op-defs -name slang)
add_public_tablegen_target(SlangOpsIncGen)

add_library(SlangDialect SHARED lib/SlangDialect.cpp lib/SlangLowering.cpp)
add_dependencies(SlangDialect SlangOpsIncGen)
target_link_libraries(SlangDialect PRIVATE MLIRIR MLIRArithDialect MLIRFuncDialect MLIRSCFDialect MLIRMemRefDialect MLIRPass MLIRTransforms)

add_executable(slang-opt tools/slang-opt/main.cpp)
target_link_libraries(slang-opt PRIVATE SlangDialect MLIROptLib MLIRArithDialect MLIRFuncDialect MLIRSCFDialect MLIRMemRefDialect)
